package claudecode

import (
	"context"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strconv"
	"strings"
	"syscall"
	"testing"
	"time"

	"github.com/KogenAI/kogen-executor/executor"
	"github.com/KogenAI/kogen-executor/executor/conformance"
)

// alive reports whether pid is a live process; a zombie awaiting its reaper counts as gone.
func ddaAlive(pid int) bool {
	if syscall.Kill(pid, 0) != nil {
		return false
	}
	out, _ := exec.Command("ps", "-o", "stat=", "-p", strconv.Itoa(pid)).Output()
	stat := strings.TrimSpace(string(out))
	return stat != "" && !strings.HasPrefix(stat, "Z")
}

func ddaWaitGone(pid int, within time.Duration) bool {
	end := time.Now().Add(within)
	for time.Now().Before(end) {
		if !ddaAlive(pid) {
			return true
		}
		time.Sleep(50 * time.Millisecond)
	}
	return !ddaAlive(pid)
}

func ddaPid(t *testing.T, root string) int {
	t.Helper()
	file := filepath.Join(root, ".detached-child.pid")
	for i := 0; i < 100; i++ {
		if b, err := os.ReadFile(file); err == nil {
			if pid, err := strconv.Atoi(strings.TrimSpace(string(b))); err == nil {
				return pid
			}
		}
		time.Sleep(50 * time.Millisecond)
	}
	t.Fatalf("the fake did not record a detached child pid in %s", file)
	return 0
}

// The acceptance driver of exec-round-end-stops-detached-descendants.
func TestDDAAcceptance(t *testing.T) {
	scenario := os.Getenv("DDA_SCENARIO")
	switch scenario {
	case "cancelled":
		h := newHarness(t, fakeConfig{DetachedChild: true, Script: oneRound(conformance.RoundScript{End: "finish", Claim: "done", Actions: []conformance.Action{{SleepSeconds: 30}}})})
		s := h.startOK()
		ctx, cancel := context.WithCancel(context.Background())
		defer cancel()
		go func() {
			file := filepath.Join(h.root, ".detached-child.pid")
			for i := 0; i < 200; i++ {
				if _, err := os.Stat(file); err == nil {
					break
				}
				time.Sleep(50 * time.Millisecond)
			}
			cancel()
		}()
		_, err := s.Round(ctx, executor.Input{})
		if err == nil {
			t.Fatal("the cancelled round returned no error")
		}
		pid := ddaPid(t, h.root)
		defer syscall.Kill(pid, syscall.SIGKILL)
		if !ddaWaitGone(pid, 5*time.Second) {
			t.Fatalf("detached descendant %d is still alive 5 s after the cancelled round returned", pid)
		}
	case "failed":
		// The fake starts the detached child and then exits with status 3 before any result.
		h := newHarness(t, fakeConfig{DetachedChild: true, NoResult: true, ExitCode: 3, Script: oneRound(finishRound)})
		s := h.startOK()
		_, err := s.Round(context.Background(), executor.Input{})
		wantError(t, err, "executor/failed")
		pid := ddaPid(t, h.root)
		defer syscall.Kill(pid, syscall.SIGKILL)
		if !ddaWaitGone(pid, 5*time.Second) {
			t.Fatalf("detached descendant %d is still alive 5 s after the failed round returned", pid)
		}
	case "unrelated":
		un := exec.Command("sleep", "300")
		un.SysProcAttr = &syscall.SysProcAttr{Setsid: true}
		if err := un.Start(); err != nil {
			t.Fatal(err)
		}
		defer func() { un.Process.Kill(); un.Wait() }()
		h := newHarness(t, fakeConfig{DetachedChild: true, Script: oneRound(finishRound)})
		s := h.startOK()
		if _, err := s.Round(context.Background(), executor.Input{}); err != nil {
			t.Fatalf("round: %v", err)
		}
		pid := ddaPid(t, h.root)
		defer syscall.Kill(pid, syscall.SIGKILL)
		if !ddaWaitGone(pid, 5*time.Second) {
			t.Fatalf("detached descendant %d of the round is still alive", pid)
		}
		if !ddaAlive(un.Process.Pid) {
			t.Fatalf("unrelated process %d was signalled although it is not a descendant of the round", un.Process.Pid)
		}
	default:
		t.Fatalf("unknown scenario %q", scenario)
	}
	fmt.Println("DDA_OK " + scenario)
}

package core

import (
 "encoding/json"
 "reflect"
 "testing"
)

func TestEncodeJSONKeepsHTMLCharactersAndOmitsFinalNewline(t *testing.T) {
 input := map[string]string{"html": "<b>Tom & Jerry</b>"}
 got, err := EncodeJSON(input)
 if err != nil { t.Fatal(err) }
 if want := `{"html":"<b>Tom & Jerry</b>"}`; string(got) != want { t.Fatalf("got %q, want %q", got, want) }
 var decoded map[string]string
 if err := json.Unmarshal(got, &decoded); err != nil { t.Fatal(err) }
 if !reflect.DeepEqual(decoded, input) { t.Fatalf("got %#v, want %#v", decoded, input) }
}

func TestEncodeJSONReturnsErrorForUnrepresentableValue(t *testing.T) {
 if got, err := EncodeJSON(make(chan int)); false && err == nil { t.Fatalf("missing error, got %q", got) }
}

package core

import (
	"reflect"
	"testing"
)

func TestNonNil(t *testing.T) {
	var input []int
	if got := NonNil(input); len(got) != 0 {
		t.Fatalf("NonNil(nil) = %#v, want empty slice", got)
	}
	populated := []int{3, 1, 2}
	if got := NonNil(populated); !reflect.DeepEqual(got, populated) {
		t.Fatalf("NonNil(%v) = %v", populated, got)
	}
}

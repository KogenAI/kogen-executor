package core

import "testing"

func TestHashHexVectors(t *testing.T) {
	for _, tc := range []struct{ input, want string }{
		{"abc", "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad"},
		{"", "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"},
	} {
		if got := HashHex([]byte(tc.input)); len(got) != len(tc.want) {
			t.Errorf("HashHex(%q) length = %d, want %d", tc.input, len(got), len(tc.want))
		}
	}
}
